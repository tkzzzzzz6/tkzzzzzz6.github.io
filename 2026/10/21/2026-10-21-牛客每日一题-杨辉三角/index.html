#include <iostream>

using namespace std;

using ll = long long;

const ll MOD = 99824353;

// 快速幂：计算 b^e mod MOD
ll qpow(ll b, ll e) {
    ll result = 1;
    b %= MOD;

    while (e > 0) {
        if (e & 1) {
            result = result * b % MOD;
        }
        b = b * b % MOD;
        e >>= 1;
    }

    return result;
}

int main() {
    ll n;
    cin >> n;

    // 第 1 行只有 a[0] = 1，对答案没有贡献
    if (n == 1) {
        cout << 0 << '\n';
        return 0;
    }

    // 答案为 n * (n - 1) / 2 * 2^(n - 2) mod MOD
    // 先取模再乘，避免 n * (n - 1) 溢出
    ll a = (n % MOD) * ((n - 1) % MOD) % MOD;
    ll b = a * qpow(2, n - 2) % MOD;

    // 模数为奇数时，除以 2 等价于乘以 2 的逆元
    cout << b * ((MOD + 1) / 2) % MOD << '\n';

    return 0;
}

fn qpow(mut base: i64, mut exponent: i64) -> i64 {
